Supabase-Einrichtung anzeigen

Lege in Supabase Auth die Benutzer an und führe im SQL Editor diese Tabelle/Policies aus:

create table if not exists public.kasse_state (
  id text primary key,
  data jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now(),
  updated_by uuid references auth.users(id)
);

alter table public.kasse_state enable row level security;

drop policy if exists "kasse_state_select" on public.kasse_state;
drop policy if exists "kasse_state_insert" on public.kasse_state;
drop policy if exists "kasse_state_update" on public.kasse_state;

create policy "kasse_state_select"
on public.kasse_state for select
to authenticated
using (id = 'shared');

create policy "kasse_state_insert"
on public.kasse_state for insert
to authenticated
with check (id = 'shared');

create policy "kasse_state_update"
on public.kasse_state for update
to authenticated
using (id = 'shared')
with check (id = 'shared');

Für eine feste Seite kannst du URL und anon key direkt im Script bei DEFAULT_SUPABASE_URL und DEFAULT_SUPABASE_ANON_KEY eintragen.
